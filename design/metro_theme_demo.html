# Ejemplo de uso
import metro_theme

def setup():
    theme = metro_theme.getTheme()  # 'dark' | 'light'
    colors = {
        'accent': '#00e5a0',
        'danger': '#ff4560',
        'value':  42,
    }
    return colors
